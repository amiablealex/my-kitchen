{# _ingredients.html — shared render helpers for recipe ingredient lines. #} {# Join amount + unit + text into a display line, dropping any part that is None or empty. A missing unit therefore renders as "0.5 avocado", never "0.5 None avocado" or a stray double space. Single source of truth for the structured (recipe_ingredients) and legacy (ingredients_json) renders. #} {% macro ingredient_line(amount, unit, text) -%} {{ [amount, unit, text] | reject('in', [none, '']) | join(' ') }} {%- endmacro %} {# Same text as ingredient_line, but with the amount wrapped in a span the servings scaler (Phase 24, servings.js) can rewrite. The recipe page only — the print/share view keeps ingredient_line, so prints stay at the recipe's own serving count. Renders identically when the scaler isn't running. #} {% macro scalable_line(amount, unit, text) -%} {%- set rest = [unit, text] | reject('in', [none, '']) | join(' ') -%} {%- if amount is not none and amount != '' -%} {{ amount }}{% if rest %} {% endif %} {%- endif -%} {{ rest }} {%- endmacro %}
